```ts
// quick-h5/src/utils/oauthClientBasic.ts
const XOR_KEY = new TextEncoder().encode('QuickBootOAuth1')

function xorBytes(bytes: Uint8Array): Uint8Array {
  const out = new Uint8Array(bytes.length)
  for (let i = 0; i < bytes.length; i++) {
    out[i] = bytes[i] ^ XOR_KEY[i % XOR_KEY.length]
  }
  return out
}

function bytesToBase64Url(bytes: Uint8Array): string {
  let binary = ''
  for (let i = 0; i < bytes.length; i++) {
    binary += String.fromCharCode(bytes[i])
  }
  // uni / 浏览器
  const b64 =
    typeof btoa === 'function'
      ? btoa(binary)
      : Buffer.from(binary, 'binary').toString('base64')
  return b64.replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/g, '')
}

export function obfuscateCredential(plain: string): string {
  return bytesToBase64Url(xorBytes(new TextEncoder().encode(plain)))
}

export function buildObfuscatedBasicAuthorization(
  clientId?: string,
  clientSecret?: string,
): string | null {
  const id = clientId ?? import.meta.env.VITE_OAUTH_CLIENT_ID ?? ''
  const secret = clientSecret ?? import.meta.env.VITE_OAUTH_CLIENT_SECRET ?? ''
  if (!id || !secret) return null
  return 'Basic ' + obfuscateCredential(`${id}:${secret}`)
}
```

- [ ] **Step 2: 实现 `http.ts`**

```ts
// quick-h5/src/api/http.ts
import { buildObfuscatedBasicAuthorization } from '@/utils/oauthClientBasic'

export type ApiResult<T> = { code: number; msg?: string; data: T }

type RequestOptions = {
  url: string
  method?: 'GET' | 'POST' | 'PUT' | 'DELETE'
  data?: unknown
  /** false 时不带 Bearer，改用 Client Basic（登录等） */
  auth?: boolean
}

function getToken(): string {
  try {
    return uni.getStorageSync('ACCESS_TOKEN') || ''
  } catch {
    return ''
  }
}

export function request<T>(options: RequestOptions): Promise<T> {
  const base = (import.meta.env.VITE_APP_BASE_API as string) || ''
  const auth = options.auth !== false
  const header: Record<string, string> = {
    'Content-Type': 'application/json;charset=utf-8',
  }
  const token = getToken()
  if (auth && token) {
    header.Authorization = 'Bearer ' + token
  } else {
    const basic = buildObfuscatedBasicAuthorization()
    if (basic) header.Authorization = basic
  }

  return new Promise((resolve, reject) => {
    uni.request({
      url: base.replace(/\/$/, '') + options.url,
      method: options.method || 'GET',
      data: options.data as AnyObject,
      header,
      success: (res) => {
        const status = res.statusCode || 0
        const body = res.data as ApiResult<T>
        if (status === 401 || body?.code === 401) {
          uni.removeStorageSync('ACCESS_TOKEN')
          uni.reLaunch({ url: '/pages/login/login' })
          reject(new Error(body?.msg || '未登录或登录已过期'))
          return
        }
        if (status < 200 || status >= 300) {
          reject(new Error(`HTTP ${status}`))
          return
        }
        if (body && typeof body.code === 'number' && body.code !== 200) {
          reject(new Error(body.msg || `业务错误 ${body.code}`))
          return
        }
        resolve(body.data)
      },
      fail: (err) => reject(new Error(err.errMsg || '网络错误')),
    })
  })
}
```

- [ ] **Step 1: user store**

```ts
// quick-h5/src/stores/user.ts
import { defineStore } from 'pinia'
import { login as loginApi, fetchMe } from '@/api/auth'

export const useUserStore = defineStore('user', {
  state: () => ({
    token: '' as string,
    userId: '' as string,
    username: '' as string,
    nickName: '' as string,
  }),
  getters: {
    isLoggedIn: (s) => !!s.token,
  },
  actions: {
    async login(username: string, password: string) {
      const tokenVo = await loginApi(username, password)
      this.token = tokenVo.accessToken
      uni.setStorageSync('ACCESS_TOKEN', tokenVo.accessToken)
      const me = await fetchMe()
      this.userId = me.userId || ''
      this.username = me.username || username
      this.nickName = me.nickName || ''
    },
    logout() {
      this.token = ''
      this.userId = ''
      this.username = ''
      this.nickName = ''
      uni.removeStorageSync('ACCESS_TOKEN')
    },
    hydrateFromStorage() {
      const t = uni.getStorageSync('ACCESS_TOKEN') || ''
      this.token = t
    },
  },
  persist: {
    paths: ['token', 'userId', 'username', 'nickName'],
  },
})
```

```vue
async function handleLogin() {
```
